   160	profile_hint() {
   161	  local kind="$1"
   162	  local profile="$2"
   163	
   164	  case "$kind" in
   165	    codex)
   166	      echo "[ai] Codex profile not found or not ready: $profile" >&2
   167	      echo "     Try: cm profile list" >&2
   168	      echo "     Try: cm profile add $profile" >&2
   169	      echo "     Try: cm run $profile" >&2
   170	      ;;
   171	    gemini)
   172	      echo "[ai] Gemini profile not found or not ready: $profile" >&2
   173	      echo "     Try: gm profile list" >&2
   174	      echo "     Try: gm profile add $profile" >&2
   175	      echo "     Try: cd ~/sb/gemini/$profile && gm run $profile" >&2
   176	      ;;
   177	    hermes)
   178	      echo "[ai] Hermes profile not found or not ready: $profile" >&2
   179	      echo "     Try: hm profile list" >&2
   180	      echo "     Try: hm profile use $profile" >&2
   181	      echo "     Try: hermes model  # or hermes setup/auth" >&2
   182	      ;;
   183	  esac

   186	failure_hint() {
   187	  local kind="$1"
   188	  local profile="$2"
   189	  local rc="$3"
   190	
   191	  echo "[ai] $kind/$profile failed with exit=$rc" >&2
   192	
   193	  case "$kind" in
   194	    codex)
   195	      echo "     Check: cd ~/sb/codex/$profile && cm task $profile \"Reply with only: OK\"" >&2
   196	      ;;
   197	    gemini)
   198	      echo "     Check: cd ~/sb/gemini/$profile && gm task $profile -- --skip-trust -p 'Reply with only: OK'" >&2
   199	      echo "     If trust error appears, run from ~/sb/gemini/$profile or re-run gm run $profile." >&2
   200	      ;;
   201	    hermes)
   202	      echo "     Check: cd ~/sb/hermes/$profile && hm task $profile \"Reply with only: OK\"" >&2
   203	      echo "     If provider error appears, run: hm profile use $profile && hermes model" >&2
   204	      echo "     If custom hgb is used, check: ai bridge status" >&2
   205	      ;;
   206	  esac

   226	run_with_timeout() {
   227	  local rc
   228	
   229	  set +e
   230	  if command -v timeout >/dev/null 2>&1 && [ -n "${AI_ASK_TIMEOUT:-}" ]; then
   231	    timeout "$AI_ASK_TIMEOUT" "$@"
   232	  else
   233	    "$@"
   234	  fi
   235	  rc=$?
   236	  set -e
   237	  return "$rc"
   238	}

   415	provider_mode_cmd() {
   416	  local kind="$1"
   417	  local mode="$2"
   418	  shift 2
   419	
   420	  local profile label workdir prompt
   421	
   422	  provider_prepare "$kind" || return $?
   423	
   424	  profile="$(provider_default_profile "$kind")" || return $?
   425	  label="$(provider_label "$kind")" || return $?
   426	
   427	  [ "$#" -gt 0 ] || {
   428	    echo "Usage: ai $mode $label [profile] PROMPT" >&2
   429	    return 2
   430	  }
   431	
   432	  # Optional explicit profile.
   433	  # Critical: check $# before touching $1.
   434	  if [ "$#" -gt 0 ] && profile_exists "$kind" "$1"; then
   435	    profile="$1"
   436	    shift
   437	  fi
   438	
   439	  profile_exists "$kind" "$profile" || {
   440	    profile_hint "$kind" "$profile"
   441	    return 127
   442	  }
   443	
   444	  [ "$#" -gt 0 ] || {
   445	    echo "Usage: ai $mode $label [profile] PROMPT" >&2
   446	    return 2
   447	  }
   448	
   449	  workdir="$(ai_workdir "$kind" "$profile")"
   450	  mkdir -p "$workdir"
   451	
   452	  if [ "${1:-}" = "-" ] || [ "${1:-}" = "--" ]; then
   453	    provider_run_manager_sandbox "$kind" "$mode" "$profile" "$workdir" "$@"
   454	    return $?
   455	  fi
   456	
   457	  prompt="$(join_prompt "$@")"
   458	  provider_run_manager_sandbox "$kind" "$mode" "$profile" "$workdir" "$prompt"
   459	}

   465	  case "${1:-}" in
   466	    codex|cm|c) echo "codex" ;;
   467	    gemini|gm|g) echo "gemini" ;;

   475	provider_mode_dispatch() {
   476	  local mode="$1"
   477	  local provider="$2"
   478	  shift 2
   479	
   480	  local kind
   481	  kind="$(provider_alias_kind "$provider")" || {
   482	    echo "ERROR: unknown provider: $provider" >&2
   483	    return 2
   484	  }
   485	
   486	  case "$mode" in
   487	    task|chat|plan)
   488	      provider_mode_cmd "$kind" "$mode" "$@"
   489	      ;;
   490	    *)
   491	      echo "ERROR: unknown provider mode: $mode" >&2
   492	      return 2
   493	      ;;
   494	  esac
   495	}

   515	default_mode() {
   516	  local mode="$1"
   517	  shift
   518	
   519	  if [ "${AI_FALLBACK_ENABLED:-true}" != "true" ]; then
   520	    provider_mode_dispatch "$mode" "$AI_DEFAULT_PROVIDER" "$@"
   521	    return $?
   522	  fi
   523	
   524	  local provider rc last_rc=1
   525	
   526	  for provider in $(provider_order); do
   527	    echo "[ai] trying provider: $provider ($mode)" >&2
   528	    provider_mode_dispatch "$mode" "$provider" "$@"
   529	    rc=$?
   530	
   531	    if [ "$rc" -eq 0 ]; then
   532	      return 0
   533	    fi
   534	
   535	    last_rc="$rc"
   536	
   537	    case "$provider" in
   538	      codex|cm|c) failure_hint codex "$AI_CODEX_PROFILE" "$rc" ;;
   539	      gemini|gm|g) failure_hint gemini "$AI_GEMINI_PROFILE" "$rc" ;;
   540	      hermes|hm|h) failure_hint hermes "$AI_HERMES_PROFILE" "$rc" ;;
   541	    esac
   542	
   543	    echo "[ai] fallback: next provider" >&2
   544	    echo >&2
   545	  done
   546	
   547	  echo "[ai] all fallback providers failed" >&2
   548	  return "$last_rc"
   549	}

   673	provider_native_passthrough() {
   674	  local kind="$1"
   675	  local mgr="$2"
   676	  local default_profile="$3"
   677	  local sub="$4"
   678	  shift 4 || true

   680	  local profile="$default_profile"

   686	      #   ai codex run lab
   687	      #   ai gemini raw tg -- --version
   688	      #
   689	      # Critical: check $# before touching $1.
   690	      if [ "$#" -gt 0 ] && profile_exists "$kind" "$1"; then
   691	        profile="$1"
   692	        shift
   693	      fi
   694	
   695	      profile_exists "$kind" "$profile" || {
   696	        profile_hint "$kind" "$profile"
   697	        return 127
   698	      }
   699	
   700	      workdir="$(ai_workdir "$kind" "$profile")"
   701	      mkdir -p "$workdir"
   702	      cd "$workdir" || return 1

   704	      if provider_is_sandbox_native_subcommand "$kind" "$sub"; then
   705	        provider_exec_sandbox "$kind" "$mgr" "$sub" "$profile" "$@"

   720	  local provider="$1"
   721	  shift
   722	
   723	  local kind mgr first mode rc default_profile
   724

   731	  default_profile="$(provider_default_profile "$kind")" || return $?

   733	  provider_prepare "$kind" || return $?
   734	
   735	  [ "$#" -gt 0 ] || exec "$mgr"
   736

   738	  mode=""
   739	
   740	  # Provider-first normalized forms.
   741	  # These MUST run before native manager passthrough.
   742	  case "$first" in
   743	    ask|chat)
   744	      mode="chat"
   745	      shift
   746	      provider_mode_dispatch "$mode" "$kind" "$@"
   747	      rc=$?
   748	      ;;
   749	    task)
   750	      mode="task"

   752	      provider_mode_dispatch "$mode" "$kind" "$@"
   753	      rc=$?
   754	      ;;
   755	    plan)
   756	      mode="plan"
   757	      shift
   758	      provider_mode_dispatch "$mode" "$kind" "$@"
   759	      rc=$?

   761	    *)

   763	        provider_native_passthrough "$kind" "$mgr" "$default_profile" "$@"
   764	      fi

   489	사용자가 "정확히", "토큰만", "한 문장만" 같은 출력 제약을 주면 그 출력 제약을 최우선으로 지켜라.

   493	task_profile() {
   494	  local p
   495	  p="$(parse_profile_first "${1:-}")"
   496	
   497	  if [ -n "${1:-}" ] && [ "$1" = "$p" ]; then
   498	    shift
   499	  fi
   500	
   501	  require_profile "$p"
   502	
   503	  local run_home
   504	  run_home="$(run_home_for_profile "$p")"
   505	  load_profile_env "$p"
   506	
   507	  # Native passthrough mode.
   508	  # Everything after -- is passed to gemini exactly.
   509	  #
   510	  # Examples:
   511	  #   gm task tg -- -p "prompt" --output-format json
   512	  #   gm task tg -- --skip-trust -p "prompt"
   513	  if [ "${1:-}" = "--" ]; then
   514	    shift
   515	    gm_run_gemini_native "$run_home" "$@"; exit "$?"
   516	  fi
   517	
   518	  # Friendly prompt mode.
   519	  #
   520	  # Examples:
   521	  #   gm task tg "prompt"
   522	  #   gm task tg -
   523	  #   cat README.md | gm task tg "Summarize this"
   524	  #
   525	  # No automatic --skip-trust here. Trust policy is Gemini-native.
   526	  if [ "$#" -eq 0 ]; then
   527	    if [ -t 0 ]; then
   528	      echo "Usage:" >&2
   529	      echo "  gm task [profile] PROMPT" >&2
   530	      echo "  gm task [profile] - < prompt.md" >&2
   531	      echo "  gm task [profile] -- <gemini native args...>" >&2
   532	      exit 2
   533	    fi
   534	    HOME="$run_home" exec gemini
   535	  fi
   536	
   537	  case "$1" in
   538	    -)
   539	      local prompt
   540	      prompt="$(cat)"
   541	      local guarded_prompt
   542	      guarded_prompt="$(gm_task_guard_prefix)
   543	
   544	$prompt"
   545	      gm_run_gemini_prompt "$run_home" "$guarded_prompt"; exit "$?"
   546	      ;;
   547	    -*)
   548	      echo "ERROR: Gemini native flags must come after --" >&2
   549	      echo >&2
   550	      echo "Examples:" >&2
   551	      echo "  gm task tg -- -p \"prompt\" --output-format json" >&2
   552	      echo "  gm task tg -- --skip-trust -p \"prompt\"" >&2
   553	      exit 2
   554	      ;;
   555	    *)
   556	      local prompt
   557	      prompt="$*"
   558	      local guarded_prompt
   559	      guarded_prompt="$(gm_task_guard_prefix)
   560	
   561	$prompt"
   562	      gm_run_gemini_prompt "$run_home" "$guarded_prompt"; exit "$?"
   563	      ;;
   564	  esac
   565	}

   346	task_profile() {
   347	  local p
   348	  p="$(parse_profile_first "${1:-}")"
   349	
   350	  if [ -n "${1:-}" ] && [ "$1" = "$p" ]; then
   351	    shift
   352	  fi
   353	
   354	  require_profile "$p"
   355	  export CODEX_HOME
   356	  CODEX_HOME="$(profile_path "$p")"
   357	
   358	  # Native passthrough mode.
   359	  #
   360	  # Everything after -- is passed to `codex exec` exactly as provided.
   361	  # No stderr suppression, no injected flags, no output filtering.
   362	  #
   363	  # Examples:
   364	  #   cm task -- --json "prompt"
   365	  #   cm task main -- --output-schema schema.json -o result.json "prompt"
   366	  if [ "${1:-}" = "--" ]; then
   367	    shift
   368	    exec codex exec "$@"
   369	  fi
   370	
   371	  # Friendly prompt mode.
   372	  #
   373	  # Manager convenience mode:
   374	  #   - adds --skip-git-repo-check
   375	  #   - hides Codex progress/banner stderr
   376	  #   - preserves final stdout
   377	  #
   378	  # Examples:
   379	  #   cm task "prompt"
   380	  #   cm task main "prompt"
   381	  #   cat prompt.md | cm task main -
   382	  local skip_git_flag
   383	  skip_git_flag="--skip-git-repo-check"
   384	
   385	  if [ "$#" -eq 0 ]; then
   386	    if [ -t 0 ]; then
   387	      echo "Usage:" >&2
   388	      echo "  cm task [profile] PROMPT" >&2
   389	      echo "  cm task [profile] - < prompt.md" >&2
   390	      echo "  cm task [profile] -- <codex exec native args...>" >&2
   391	      exit 2
   392	    fi
   393	    codex exec "$skip_git_flag" - 2>/dev/null
   394	    exit $?
   395	  fi
   396	
   397	  case "$1" in
   398	    -)
   399	      codex exec "$skip_git_flag" - 2>/dev/null
   400	      exit $?
   401	      ;;
   402	    -*)
   403	      echo "ERROR: Codex native flags must come after --" >&2
   404	      echo >&2
   405	      echo "Examples:" >&2
   406	      echo "  cm task -- --json \"prompt\"" >&2
   407	      echo "  cm task main -- --skip-git-repo-check --json \"prompt\"" >&2
   408	      echo "  cm task main -- --output-schema schema.json -o result.json \"prompt\"" >&2
   409	      exit 2
   410	      ;;
   411	    *)
   412	      local prompt
   413	      prompt="$*"
   414	      codex exec "$skip_git_flag" "$prompt" 2>/dev/null
   415	      exit $?
   416	      ;;
   417	  esac
   418	}

    86	  hm task main -- -z "Reply with only: OK"
    87	  hm task main -- --ignore-rules -z "Reply with only: OK"
    88	
    89	CWD policy:
    90	  hm sb <command> <profile> ...                 # force ~/sb/hermes/<profile>
    91	  hm sandbox <command> <profile> ...            # same as sb
    92	  HM_CWD_TTY_POLICY=ask|cwd|sandbox
    93	  HM_CWD_NONTTY_POLICY=cwd|sandbox
    94	
    95	Notes:
    96	  ask/chat = no-tool answer mode.
    97	  task     = tool-capable mode.
    98	  plan     = plan-only mode.
    99	  raw      = exact native Hermes passthrough.
   100	  gateway  = hgw passthrough.
   101	USAGE
   102	}

   123	run_profile() {
   124	  local p
   125	  p="$(parse_profile_first "${1:-}")"
   126	
   127	  if [ -n "$p" ]; then
   128	    shift
   129	    use_if_profile "$p"
   130	    echo "Hermes profile active: $p"
   131	    echo
   132	  fi

   135	    shift

   141	task_profile() {
   142	  local p
   143	  p="$(parse_profile_first "${1:-}")"
   144	
   145	  if [ -n "$p" ]; then
   146	    shift
   147	    use_if_profile "$p"
   148	  fi
   149	
   150	  # Native passthrough mode:
   151	  #   hm task main -- -z "prompt"
   152	  #   hm task main -- --ignore-rules -z "prompt"
   153	  # Everything after -- is passed to hermes exactly.
   154	  if [ "${1:-}" = "--" ]; then
   155	    shift
   156	    exec hermes "$@"
   157	  fi
   158	
   159	  # Friendly prompt mode:
   160	  #   hm task main "prompt"
   161	  #   hm task main -
   162	  if [ "$#" -eq 0 ]; then
   163	    if [ -t 0 ]; then
   164	      echo "Usage:" >&2
   165	      echo "  hm task [profile] PROMPT" >&2
   166	      echo "  hm task [profile] - < prompt.md" >&2
   167	      echo "  hm task [profile] -- <hermes native args...>" >&2
   168	      exit 2
   169	    fi
   170	    local stdin_prompt
   171	    stdin_prompt="$(cat)"
   172	    exec hermes -z "$stdin_prompt"
   173	  fi
   174	
   175	  case "$1" in
   176	    -)
   177	      local prompt
   178	      prompt="$(cat)"
   179	      exec hermes -z "$prompt"
   180	      ;;
   181	    -*)
   182	      echo "ERROR: Hermes native flags must come after --" >&2
   183	      echo >&2
   184	      echo "Examples:" >&2
   185	      echo "  hm task main -- -z \"prompt\"" >&2
   186	      echo "  hm task main -- --ignore-rules -z \"prompt\"" >&2
   187	      exit 2
   188	      ;;
   189	    *)
   190	      local prompt
   191	      prompt="$*"
   192	      exec hermes -z "$prompt"
   193	      ;;
   194	  esac
   195	}

   362	  local val="${2:-}"
   363	
   364	  if [ -z "$key" ] || [ -z "$val" ]; then
   365	    echo "Usage: hgw bridge set profile|model|port|host|public-model|session|profiles VALUE" >&2
   366	    exit 2

   369	  mkdir -p "$(dirname "$BRIDGE_ENV")"
   370	  touch "$BRIDGE_ENV"

   387	    python - "$BRIDGE_ENV" "$key" "$val" <<'PY'
   388	from pathlib import Path
   389	import sys
   390	p = Path(sys.argv[1])
   391	key = sys.argv[2]
   392	val = sys.argv[3]
   393	lines = p.read_text().splitlines()
   394	out = []
   395	for line in lines:
   396	    if line.startswith(key + "="):
   397	        out.append(f'{key}="{val}"')
   398	    else:
   399	        out.append(line)
   400	p.write_text("\n".join(out) + "\n")
   401	PY
   402	  else
   403	    printf '%s="%s"\n' "$key" "$val" >> "$BRIDGE_ENV"
   404	  fi
   405	
   406	  echo "set $key=$val"
   407	}

    83	def guard_gemini_prompt_text(text: str) -> str:
    84	    """Sanitize Hermes/Gateway text before handing it to Gemini CLI.
    85	
    86	    B안:
    87	      Gemini CLI treats @path-like text as a file reference. Break only
    88	      file-ish @ references such as @tests/foo.py, @./x, @../x, @/x, @~/x.
    89	
    90	    C안:
    91	      Hermes skill catalog / referenced-file blocks are useful for Hermes
    92	      itself, but they pollute Gemini CLI prompts and can trigger @file import.
    93	    """
    94	    if not isinstance(text, str) or not text:
    95	        return text
    96	
    97	    text = _HGB_HERMES_SKILLS_RE.sub(
    98	        '<available_skills omitted by hgb gemini prompt guard>',
    99	        text,
   100	    )
   101	
   102	    text = _HGB_REFERENCED_FILES_RE.sub(
   103	        '\\n[referenced file contents omitted by hgb gemini prompt guard]\\n',
   104	        text,
   105	    )
   106	
   107	    # Insert zero-width space after @ only for file/path-looking references.
   108	    text = _HGB_AT_PATH_RE.sub('@\u200b', text)
   109	    return text

   350	def cleanup_expired(silent=False):
   351	    ensure_dirs()
   352	    moved = 0
   353	    t = now()
   354	
   355	    for p in list(PENDING.glob("*.json")):
   356	        try:
   357	            req = json.loads(p.read_text(encoding="utf-8"))
   358	        except Exception:
   359	            continue
   360	
   361	        if int(req.get("expires_at", 0)) < t:
   362	            req["status"] = "expired"
   363	            req["expired_at"] = t
   364	            save_req(req, EXPIRED)
   365	            p.unlink(missing_ok=True)
   366	            moved += 1
   367	
   368	    if not silent:
   369	        print(f"expired moved: {moved}")
   370	
   371	
   372	def cmd_cleanup(args):
   373	    cleanup_expired(silent=False)
   374	
   375	    cutoff = now() - int(args.keep_seconds)
   376	    removed = 0
   377	
   378	    for folder in [DONE, REJECTED, EXPIRED]:
   379	        for p in list(folder.glob("*.json")):
   380	            try:
   381	                req = json.loads(p.read_text(encoding="utf-8"))
   382	                ts = int(req.get("done_at") or req.get("rejected_at") or req.get("expired_at") or req.get("created_at") or 0)
   383	            except Exception:
   384	                ts = 0
   385	            if ts and ts < cutoff:
   386	                p.unlink(missing_ok=True)
   387	                removed += 1
   388	
   389	    print(f"old removed: {removed}")
